🔗 Cấu Hình Webhook Google Sheet

Để đồng bộ tự động 2 phút/lần lên Google Sheet của bạn, hãy nhập URL Webhook (Google Apps Script Web App URL) vào ô bên dưới:

📋 Hướng Dẫn 3 Bước Tạo Webhook Trên Google Sheets (Chỉ mất 2 phút)

  1. Mở Google Sheet của bạn → Chọn menu Tiện ích mở rộng (Extensions) → Chọn Apps Script.
  2. Xóa toàn bộ mã cũ và dán đoạn mã Google Apps Script mẫu bên dưới vào.
  3. Bấm nút Triển khai (Deploy) → Chọn Tùy chọn triển khai mới (New deployment):
    • Chọn loại: Ứng dụng web (Web app)
    • Ai có quyền truy cập (Who has access): Bất kỳ ai (Anyone)
  4. Copy URL web app vừa nhận được và dán vào ô bên trên → Bấm Lưu Cấu Hình!
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var payload = JSON.parse(e.postData.contents);
    var spins = payload.data || [];
    
    // Nếu trang tính còn trống, tạo dòng tiêu đề (Header)
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "ID", "Thời Gian", "Mã Đại Lý", "Tên Đại Lý", "Tỉnh Thành", 
        "Chủ Đại Lý", "Số Điện Thoại", "Địa Chỉ", "Mã Dự Thưởng", 
        "Số Serial", "Phần Quà Trúng", "Thời Gian Đồng Bộ"
      ]);
    }
    
    // Thêm các dòng dữ liệu quay thưởng mới
    for (var i = 0; i < spins.length; i++) {
      var s = spins[i];
      sheet.appendRow([
        s.id, s.spin_time, s.agency_code, s.agency_name, s.province,
        s.owner_name, s.phone, s.address, s.entry_code,
        s.serial_number, s.prize_name, new Date()
      ]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: spins.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
